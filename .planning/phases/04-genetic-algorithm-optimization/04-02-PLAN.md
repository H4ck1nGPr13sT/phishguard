python3 -c "
import joblib
from sklearn.metrics import f1_score
from pathlib import Path

# Load test data
data = joblib.load('cache/url_training_data.joblib')
X_test, y_test = data['X_test'], data['y_test']

classifiers = ['rf', 'svm', 'mlp', 'xgb', 'lr', 'nb', 'dt']
improvements = []

print('Baseline vs Optimized F1 Comparison:')
print('-' * 50)

for name in classifiers:
    # Load baseline
    baseline_path = Path(f'models/{name}_pipeline.joblib')
    optimized_path = Path(f'models/optimized/{name}_optimized.joblib')

    baseline = joblib.load(baseline_path)
    optimized = joblib.load(optimized_path)

    # Evaluate both
    baseline_f1 = f1_score(y_test, baseline.predict(X_test))
    optimized_f1 = f1_score(y_test, optimized.predict(X_test))
    delta = optimized_f1 - baseline_f1
    improvements.append(delta)

    status = 'OK' if optimized_f1 >= baseline_f1 - 0.01 else 'REGRESSION'
    print(f'{name}: baseline={baseline_f1:.4f}, optimized={optimized_f1:.4f}, delta={delta:+.4f} [{status}]')

# Summary
positive = sum(1 for d in improvements if d > 0)
print('-' * 50)
print(f'Improved: {positive}/7 classifiers')
print(f'Average delta: {sum(improvements)/len(improvements):+.4f}')

# Assert no significant regressions (within 1% tolerance)
for name, delta in zip(classifiers, improvements):
    assert delta >= -0.01, f'{name} has significant regression: {delta:.4f}'
print('All optimized models verified (no regressions beyond 1% tolerance)')
"
