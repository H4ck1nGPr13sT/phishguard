```python
def extract_all_features(samples, content_type: ContentType):
    features = []
    labels = []
    for sample in samples:
        if content_type == ContentType.EMAIL:
            feats = extract_email_features(sample["content"].encode())
        else:
            feats = extract_sms_features(sample["content"])
        features.append(list(feats.values()))
        labels.append(sample["label"])
    return np.array(features), np.array(labels)
```

def train_classifier(name, clf, X_train, y_train, X_test, y_test):
    pipeline = Pipeline([
        ("scaler", StandardScaler()),
        ("classifier", clf)
    ])
    pipeline.fit(X_train, y_train)
    accuracy = pipeline.score(X_test, y_test)
    print(f"{name}: {accuracy:.4f} accuracy")
    return pipeline
