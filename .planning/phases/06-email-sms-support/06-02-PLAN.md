```python
"""NLP-based text feature extraction for email/SMS phishing detection.

This module extracts linguistic features from text content:
- Lexical features (character counts, word patterns)
- Syntactic features (POS tag distributions, sentence structure)
- Stylometric features (readability, complexity)
- Sentiment features (urgency, threat indicators)
"""

import re
from typing import Dict
import spacy
import textstat

class TextFeatureExtractor:
    """Extract NLP features from text content.

    Load spaCy model ONCE at initialization, not per extraction.
    Disable NER (not needed) for faster processing.
    """

    def __init__(self):
        # Load spaCy model once (disable NER for speed)
        self.nlp = spacy.load("en_core_web_sm", disable=["ner"])

    def extract_all_features(self, text: str) -> Dict[str, float]:
        """Extract all NLP features from text.

        Returns ~50 features combining lexical, syntactic, stylometric, sentiment.
        """
        if not text or not text.strip():
            return _get_default_text_features()

        features = {}
        features.update(self.extract_lexical_features(text))
        features.update(self.extract_syntactic_features(text))
        features.update(self.extract_stylometric_features(text))
        features.update(self.extract_sentiment_features(text))
        return features
```

**extract_lexical_features(text) -> Dict** (~15 features):
- text_length: len(text)
- word_count: len(text.split())
- avg_word_length: mean of word lengths
- digit_count: sum of digits
- digit_ratio: digit_count / text_length
- uppercase_count: sum of uppercase chars
- uppercase_ratio: uppercase_count / text_length
- exclamation_count: text.count("!")
- question_count: text.count("?")
- dollar_count: text.count("$")
- url_count: count of http/https patterns
- email_count: count of @domain patterns
- phone_pattern_count: count of phone-like patterns
- special_char_ratio: non-alphanumeric / total
- whitespace_ratio: whitespace / total

**extract_syntactic_features(text) -> Dict** (~15 features):
- Use self.nlp(text) to get spaCy Doc
- sentence_count: len(list(doc.sents))
- avg_sentence_length: tokens / sentences
- token_count: len(doc)
- verb_count, verb_ratio: VERB POS tags
- noun_count, noun_ratio: NOUN POS tags
- pronoun_count, pronoun_ratio: PRON POS tags
- adjective_count, adjective_ratio: ADJ POS tags
- adverb_count, adverb_ratio: ADV POS tags
- punct_count: PUNCT tokens
- imperative_verb_count: verbs at sentence start (indicative of commands)

**extract_stylometric_features(text) -> Dict** (~10 features):
- flesch_reading_ease: textstat.flesch_reading_ease(text)
- flesch_kincaid_grade: textstat.flesch_kincaid_grade(text)
- gunning_fog: textstat.gunning_fog(text)
- smog_index: textstat.smog_index(text)
- coleman_liau_index: textstat.coleman_liau_index(text)
- automated_readability_index: textstat.automated_readability_index(text)
- syllable_count: textstat.syllable_count(text)
- lexicon_count: textstat.lexicon_count(text, removepunct=True)
- difficult_words: textstat.difficult_words(text)
- lexical_diversity: unique_words / total_words

**extract_sentiment_features(text) -> Dict** (~10 features):
Phishing-specific urgency and threat indicators:
- urgency_keyword_count: count of ["urgent", "immediately", "act now", "asap", "expire", "suspended", "verify", "confirm"]
- threat_keyword_count: count of ["suspended", "terminated", "locked", "unauthorized", "fraud", "security"]
- action_keyword_count: count of ["click", "login", "update", "verify", "confirm", "download"]
- reward_keyword_count: count of ["won", "winner", "prize", "free", "claim", "reward"]
- has_urgency: 1 if urgency_keyword_count > 0
- has_threat: 1 if threat_keyword_count > 0
- has_action_request: 1 if action_keyword_count > 0
- has_reward_claim: 1 if reward_keyword_count > 0
- personal_pronoun_ratio: ("you", "your") / word_count
- impersonal_greeting: 1 if starts with "Dear Customer", "Dear User", etc.

CRITICAL: Do NOT set stop_words in any tokenizer - words like "your", "now", "urgent" are phishing indicators per research.

**_get_default_text_features() -> Dict** - All zeros/defaults

**extract_text_features(text: str, extractor: TextFeatureExtractor) -> Dict** - Module-level convenience function
