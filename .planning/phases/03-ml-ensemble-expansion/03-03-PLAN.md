1. Create src/models/disagreement.py:

   Imports:
   - numpy as np
   - scipy.stats.entropy

   Constants:
   - DISAGREEMENT_THRESHOLD = 0.7  # Configurable threshold for edge case detection

   Functions:

   a) calculate_disagreement(individual_predictions: dict) -> float:
      """Calculate normalized Shannon entropy across classifier predictions.

      Args:
          individual_predictions: Dict from get_individual_predictions()
              {'rf': {'phishing_probability': 0.8, ...}, 'svm': {...}, ...}

      Returns:
          float: Disagreement score 0-1 where:
              0 = all classifiers agree (unanimous vote)
              1 = maximum disagreement (even split)
      """
      - Extract predictions (0 or 1 based on >0.5 threshold) from all classifiers
      - Count votes for each class using np.unique(predictions, return_counts=True)
      - Normalize to probability distribution: pk = counts / total
      - Calculate Shannon entropy: H = entropy(pk, base=2)
      - Normalize by max entropy: max_H = np.log2(n_classifiers)
      - Return H / max_H if max_H > 0 else 0.0

      EDGE CASE: All classifiers agree -> entropy = 0 -> return 0.0
      EDGE CASE: Near-even split (3-4 or 4-3) -> high entropy -> return ~0.98

   b) is_edge_case(disagreement_score: float, threshold: float = DISAGREEMENT_THRESHOLD) -> bool:
      """Check if disagreement score indicates an edge case."""
      return disagreement_score > threshold

   c) get_disagreement_summary(individual_predictions: dict) -> dict:
      """Get detailed disagreement analysis.

      Returns dict with:
          - score: normalized entropy (0-1)
          - is_edge_case: bool
          - vote_distribution: {'phishing': N, 'legitimate': M}
          - agreeing_classifiers: list of classifier names that agree with majority
          - dissenting_classifiers: list of classifier names that disagree
      """

3. Update src/api/endpoints.py - add ensemble endpoint:

   @router.post("/predict/ensemble", response_model=EnsemblePredictionResponse, tags=["prediction"])
   def predict_ensemble(request: URLRequest):
       """
       Predict phishing using ensemble of 7 classifiers with disagreement detection.

       Returns individual predictions from all classifiers plus ensemble verdict.
       Flags edge cases when classifier disagreement exceeds threshold.

       Requirements addressed:
       - ENS-05: Reports disagreement between classifiers
       - WEB-05: Returns comparison of all classifier results
       """
       start_time = time.time()

       if "voting_soft" not in ml_models:
           raise HTTPException(status_code=503, detail="Ensemble models not loaded")

       # Extract features (same as /predict)
       features = extract_url_features(request.url)
       feature_array = np.array([list(features.values())])

       # Get individual predictions from voting ensemble
       individual_preds = get_individual_predictions(ml_models["voting_soft"], feature_array)

       # Get ensemble prediction
       ensemble_proba = ml_models["voting_soft"].predict_proba(feature_array)[0]

       # Calculate disagreement
       disagreement_info = get_disagreement_summary(individual_preds)

       processing_time = (time.time() - start_time) * 1000

       return EnsemblePredictionResponse(
           url=request.url,
           ensemble_prediction="phishing" if ensemble_proba[1] > 0.5 else "legitimate",
           ensemble_probability=float(ensemble_proba[1]),
           ensemble_confidence=float(max(ensemble_proba)),
           individual_predictions=[
               ClassifierResult(name=name, **pred) for name, pred in individual_preds.items()
           ],
           disagreement=DisagreementInfo(**disagreement_info),
           voting_method="soft",
           processing_time_ms=processing_time
       )
